<Errors
        model={`${modelPath}.driver.dateOfBirth`}
        messages={{
          minDateError: config.minDateError,
          maxDateError: config.maxDateError,
          invalidDate: config.invalidDate
        }}
        show={true}
        component={({ fieldValue: { errors }, children: message }) => invalidDateErrorMessage(errors, message)}
      />

const invalidDateErrorMessage = (errors: any, message: any) => {
    if (errors.invalidDate) {
      return (
        <div className="error-container">
          <span>{message}</span>
        </div>
      );
    }
    return (
      <div className="error-container">
        <span>{message}</span>
      </div>
    );
  };

}
