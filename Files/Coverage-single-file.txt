const coverage = require('./coverage/coverage-final.json');

const file = Object.values(coverage).find(
  item => item.path.endsWith('src/components/Page1.tsx')
);

if (!file) {
  console.error('File not found');
  process.exit(1);
}

const percentage = values => {
  const total = values.length;
  const covered = values.filter(v => v > 0).length;
  return total === 0 ? 100 : (covered / total) * 100;
};

const branchValues = Object.values(file.b).flat();

console.log({
  file: file.path,
  statements: `${percentage(Object.values(file.s)).toFixed(2)}%`,
  functions: `${percentage(Object.values(file.f)).toFixed(2)}%`,
  branches: `${percentage(branchValues).toFixed(2)}%`,
});
